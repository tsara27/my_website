Spreadsheet Setup Wizard

Because this metric web tracker runs purely in your web browser, it securely communicates with your Google Spreadsheet through a Google Apps Script Web App. Follow these 5 steps to create one for free:

  1. 1

    Open your Google Sheet

    Open your Weight Tracker Sheet. Log in with your active Google account.

  2. 2

    Open Google Apps Script

    In your spreadsheet's top menu bar, click on Extensions > Apps Script. This will open an editor window.

  3. 3

    Paste the Web Connector Code

    Delete any code in the editor, and paste the code snippet below: 

    function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([]))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = data[i][j];
    }
    rows.push(row);
  }
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var params = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Create headers if empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Date", "Weight (kg)", "Body Fat (%)", "Visceral Fat", "BMI", "Notes"]);
    }
    
    // Check if params is an array of entries (bulk upload)
    if (Array.isArray(params)) {
      var data = sheet.getDataRange().getValues();
      var dateToRowIndex = {};
      for (var i = 1; i < data.length; i++) {
        var sheetDate = data[i][0];
        if (sheetDate instanceof Date) {
          var tzOffset = sheetDate.getTimezoneOffset() * 60000;
          sheetDate = new Date(sheetDate.getTime() - tzOffset).toISOString().split('T')[0];
        } else {
          sheetDate = String(sheetDate).trim();
        }
        dateToRowIndex[sheetDate] = i + 1;
      }
      
      for (var k = 0; k < params.length; k++) {
        var p = params[k];
        var targetDate = p.Date;
        var rowData = [
          p.Date,
          parseFloat(p.Weight),
          parseFloat(p.BodyFat),
          parseFloat(p.VisceralFat),
          parseFloat(p.BMI || 0),
          p.Notes || ""
        ];
        var rIdx = dateToRowIndex[targetDate];
        if (rIdx) {
          sheet.getRange(rIdx, 1, 1, rowData.length).setValues([rowData]);
        } else {
          sheet.appendRow(rowData);
          dateToRowIndex[targetDate] = sheet.getLastRow();
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Successfully synced " + params.length + " entries to sheet" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Process single entry
    var data = sheet.getDataRange().getValues();
    var dateIndex = -1;
    var targetDate = params.Date; // Format YYYY-MM-DD
    
    for (var i = 1; i < data.length; i++) {
      var sheetDate = data[i][0];
      if (sheetDate instanceof Date) {
        var tzOffset = sheetDate.getTimezoneOffset() * 60000;
        sheetDate = new Date(sheetDate.getTime() - tzOffset).toISOString().split('T')[0];
      } else {
        sheetDate = String(sheetDate).trim();
      }
      if (sheetDate === targetDate) {
        dateIndex = i + 1;
        break;
      }
    }
    
    var rowData = [
      params.Date,
      parseFloat(params.Weight),
      parseFloat(params.BodyFat),
      parseFloat(params.VisceralFat),
      parseFloat(params.BMI || 0),
      params.Notes || ""
    ];
    
    if (dateIndex !== -1) {
      sheet.getRange(dateIndex, 1, 1, rowData.length).setValues([rowData]);
    } else {
      sheet.appendRow(rowData);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Successfully saved to sheet" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
